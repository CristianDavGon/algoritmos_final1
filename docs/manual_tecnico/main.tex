\newcommand{\figura}[4]{%
  \begin{figure}[H]
    \centering
    \IfFileExists{figures/#1}{%
      \includegraphics[width=#2\textwidth]{figures/#1}%
    }{%
      \fbox{\begin{minipage}{#2\textwidth}%
        \centering\vspace{1.8cm}%
        \textcolor{gray}{\small\textit{Diagrama pendiente --- agregar archivo en figures/}}%
        \vspace{1.8cm}%
      \end{minipage}}%
    }
    \caption{#3}
    \label{fig:#4}
  \end{figure}%
}
